        // CORS SAFE FETCH
        window.smartFetch = async function(url) {
            const proxies = [
                (u) => u,
                (u) => `https://corsproxy.io/?${encodeURIComponent(u)}`,
                (u) => `https://api.allorigins.win/raw?url=${encodeURIComponent(u)}`
            ];

            for (let p of proxies) {
                try {
                    const res = await fetch(p(url));
                    if (res.ok) {
                        return await res.text();
                    }
                } catch(e) {}
            }
            throw new Error("No se pudo obtener el recurso mediante CORS Proxy");
        };

        // PARSE M3U TEXT
        window.parseM3uText = function(text, sourceName = "M3U List") {
            const lines = text.split('\n');
            const newChannels = [];
            let currentChannel = null;

            lines.forEach(line => {
                line = line.trim();
                if (line.startsWith('#EXTINF:')) {
                    currentChannel = { id: 'ch_' + Math.random().toString(36).substr(2, 9), source: sourceName };
                    
                    const tvgIdMatch = line.match(/tvg-id="([^"]*)"/);
                    if (tvgIdMatch) currentChannel.tvgId = tvgIdMatch[1];

                    const logoMatch = line.match(/tvg-logo="([^"]*)"/);
                    if (logoMatch) currentChannel.logo = logoMatch[1];

                    const groupMatch = line.match(/group-title="([^"]*)"/);
                    if (groupMatch) currentChannel.category = groupMatch[1];

                    const nameParts = line.split(',');
                    currentChannel.name = nameParts[nameParts.length - 1].trim();
                } else if (line.length > 0 && !line.startsWith('#')) {
                    if (currentChannel) {
                        currentChannel.url = line;
                        if (!currentChannel.category) currentChannel.category = "General";
                        newChannels.push(currentChannel);
                        currentChannel = null;
                    }
                }
            });

            window.masterChannelList.push(...newChannels);
            window.updateCategoriesSelect();
            window.renderChannels();
            return newChannels.length;
        };
